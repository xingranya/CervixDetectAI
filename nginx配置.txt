server {
    listen 9001;
    server_name 36.50.226.32;
    
    # 前端静态文件根目录
    root /www/wwwroot/36.50.226.32_9001;
    index index.html;
    
    # 启用 Gzip 压缩
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/javascript application/json application/xml+rss image/svg+xml;
    
    # 🔥 重要！上传文件访问（放在前面，优先匹配）
    location /uploads/ {
        alias /www/wwwroot/CervixDetectAI/server/uploads/;
        expires 30d;
        add_header Cache-Control "public, immutable";
        add_header Access-Control-Allow-Origin *;
        access_log off;
    }
    
    # 报告文件访问
    location /reports/ {
        alias /www/wwwroot/CervixDetectAI/server/reports/;
        expires 7d;
    }
    
    # API 反向代理
  # API请求代理到后端
    location /api/ {
        proxy_pass http://127.0.0.1:3000/api/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
    }
    
    location /fonts/ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # 前端路由
    location / {
        root /www/wwwroot/36.50.226.32_9001;
        try_files $uri $uri/ /index.html;
        add_header Cache-Control "no-cache, no-store, must-revalidate";
    }
    
    # 🔥 修改这里：只匹配前端静态资源，排除 /uploads/
    location ~* ^(?!/uploads/).*\.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }
    
    # 安全头
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    
    # 禁止访问隐藏文件
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }
    
    # 访问日志
    access_log /www/wwwlogs/36.50.226.32_9001.log;
    error_log /www/wwwlogs/36.50.226.32_9001.error.log;
}
